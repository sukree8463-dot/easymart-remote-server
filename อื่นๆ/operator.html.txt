    <!-- โหลด Socket.io จาก CDN -->
    <script src="https://cdn.socket.io/4.7.2/socket.io.min.js"></script>
    <script>
        const socket = io("http://localhost:3000");

        socket.on("connect", () => {
            console.log("Operator Connected to Server ID:", socket.id);
        });

        function connectToClient() {
            const clientId = document.getElementById("clientIdInput").value.trim();
            if (!clientId) {
                alert("กรุณากรอก Client ID ก่อนครับ");
                return;
            }

            document.getElementById("statusText").innerText = "สถานะ: กำลังส่งคำขอเชื่อมต่อไปยัง " + clientId + "...";
            
            // ส่งคำขอไปยัง Server เพื่อบอกให้ Client เริ่มสตรีมภาพ
            socket.emit("request-stream", { clientId: clientId });
        }

        // รับภาพหน้าจอเฟรมต่อเฟรมจาก Client
        socket.on("screen-frame", (data) => {
            document.getElementById("statusText").innerText = "สถานะ: กำลังสตรีมหน้าจออยู่...";
            const imgElement = document.getElementById("screen-view");
            imgElement.src = "data:image/jpeg;base64," + data.image;
        });
    </script>
